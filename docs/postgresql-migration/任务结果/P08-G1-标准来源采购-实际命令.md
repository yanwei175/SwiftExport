# P08-G1 标准来源采购实际进程命令

主控只执行一个预登记真实PG场景；凭据从当前用户配置读入内存，未输出或写入文档。库名/门禁显式指定，原环境值finally恢复。实际首轮exit0/1通过，未重跑。

```powershell
$ErrorActionPreference='Stop'
$taskSettingsDoc=Get-Content SwiftExport.WebApi/appsettings.json -Raw | ConvertFrom-Json
$taskPgSettings=[System.Data.Common.DbConnectionStringBuilder]::new()
$taskPgSettings.set_ConnectionString([string]$taskSettingsDoc.ConnectionStrings.Default)
if([string]$taskPgSettings.get_Item('Host') -ne '192.168.3.12' -or [string]$taskPgSettings.get_Item('Database') -ne 'dv_next'){throw '当前配置不是指定实例运行库'}
$taskPgSettings.set_Item('Database','dv_next_pg_failure_tests_20261004')
$taskNames=@('SWIFTEXPORT_RUN_PG_FAILURE_TESTS','SWIFTEXPORT_PG_FAILURE_TEST_CONNECTION','SWIFTEXPORT_PG_RUNTIME_STANDARD_SOURCE_BATCH')
$taskPrevious=@{}
foreach($taskName in $taskNames){$taskPrevious[$taskName]=[Environment]::GetEnvironmentVariable($taskName)}
try{
 [Environment]::SetEnvironmentVariable('SWIFTEXPORT_RUN_PG_FAILURE_TESTS','1')
 [Environment]::SetEnvironmentVariable('SWIFTEXPORT_PG_FAILURE_TEST_CONNECTION',$taskPgSettings.get_ConnectionString())
 [Environment]::SetEnvironmentVariable('SWIFTEXPORT_PG_RUNTIME_STANDARD_SOURCE_BATCH','25df913f-e4f9-4082-99ad-782a3d3eed44')
 dotnet test SwiftExport.Application.Tests/SwiftExport.Application.Tests.csproj --no-build --no-restore --filter 'FullyQualifiedName~RealPg_RuntimeStandardSourceCreatesAndAppendsPurchase' --results-directory TestResults/postgresql-migration --logger 'trx;LogFileName=p08-g1-source-purchase-first.trx' -v minimal
 $taskExit=$LASTEXITCODE
}finally{
 foreach($taskName in $taskNames){[Environment]::SetEnvironmentVariable($taskName,$taskPrevious[$taskName])}
 $taskSettingsDoc=$null
 $taskPgSettings=$null
}
exit $taskExit
```
